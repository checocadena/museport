__global__ void residual_forward_kernel(
    float* out, float* inp1, float* inp2, int N) {
  int idx = blockIdx.x * blockDim.x + threadIdx.x;
  if (idx < N) {
    out[idx] = __ldcs(&inp1[idx])
             + __ldcs(&inp2[idx]);
  }
}

static void residual_forward(
    float *out, const float *a,
    const float *b, size_t N) {
  launch_1d(N, [=](sycl::id<1> id) {
    out[id[0]] = a[id[0]] + b[id[0]];
  });
}
